# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). The forests (the Faydarks,
# Kithicor, Nektulos...): the lawn with Kithicor's ferns and small pines among it (oldkithicor.eqg).
donor oldkithicor.eqg
max_slope 45
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 1.7 +greenish lawn fern:0.5 fern02:0.5 fernA:0.25 fern02A:0.25 smallpine:0.15
ground *grass* 1.7 +greenish lawn fern:0.5 fern02:0.5 fernA:0.25 fern02A:0.25 smallpine:0.15
ground *gras* 1.7 +greenish lawn fern:0.5 fern02:0.5 fernA:0.25 fern02A:0.25 smallpine:0.15
include stone
# Anything else that is clearly green.
ground * 1.7 +green lawn fern:0.5 fern02:0.5 fernA:0.25 fern02A:0.25 smallpine:0.15
