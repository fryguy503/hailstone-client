# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). The Plane of Growth: Arcstone's
# tall flowering grass and ferns (arcstone.eqg) with the Moors' flowering grass (moors.eqg) among the lawn.
donor arcstone.eqg
donor moors.eqg
max_slope 45
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 0.95 +greenish lawn arcstone/grass1:1 arcstone/grass4:0.4 jungle1:0.4 grasslandflora:0.5 grasslandplant2:0.3
ground *grass* 0.95 +greenish lawn arcstone/grass1:1 arcstone/grass4:0.4 jungle1:0.4 grasslandflora:0.5 grasslandplant2:0.3
ground *gras* 0.95 +greenish lawn arcstone/grass1:1 arcstone/grass4:0.4 jungle1:0.4 grasslandflora:0.5 grasslandplant2:0.3
include stone
# Anything else that is clearly green.
ground * 0.95 +green lawn arcstone/grass1:1 arcstone/grass4:0.4 jungle1:0.4 grasslandflora:0.5 grasslandplant2:0.3
