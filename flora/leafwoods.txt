# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). The leafy woods (the Feerrott):
# the lawn with the revamped Feerrott's leafy bushes and ferns (feerrott2.eqg).
donor feerrott2.eqg
max_slope 45
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 1.51 +greenish lawn feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
ground *grass* 1.51 +greenish lawn feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
ground *gras* 1.51 +greenish lawn feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
include stone
# Anything else that is clearly green.
ground * 1.51 +green lawn feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
