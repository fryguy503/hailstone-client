# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). Luclin's open ground (the Twilight
# Sea, the Grimling Forest, Hollowshade...): the lawn with Misty Thicket's shrubs and plants
# (mistythicket.eqg).
donor mistythicket.eqg
max_slope 45
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 0.61 +greenish lawn ShadeGrass:0.8 Plant:0.4 Shrub1:0.3 Shrub2:0.3
ground *grass* 0.61 +greenish lawn ShadeGrass:0.8 Plant:0.4 Shrub1:0.3 Shrub2:0.3
ground *gras* 0.61 +greenish lawn ShadeGrass:0.8 Plant:0.4 Shrub1:0.3 Shrub2:0.3
include stone
# Anything else that is clearly green.
ground * 0.61 +green lawn ShadeGrass:0.8 Plant:0.4 Shrub1:0.3 Shrub2:0.3
