# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). The grasslands (the Karanas, the
# Commonlands, Lake Rathetear): the lawn with the Moors' flowering grass and wheat (moors.eqg).
donor moors.eqg
max_slope 45
# The grasslands' lawn grows taller.
lawn_height 2.4 4
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 0.51 +greenish lawn grasslandflora:0.6 grasslandplant2:0.4 grasslandgrass:0.4 wheat:0.25
ground *grass* 0.51 +greenish lawn grasslandflora:0.6 grasslandplant2:0.4 grasslandgrass:0.4 wheat:0.25
ground *gras* 0.51 +greenish lawn grasslandflora:0.6 grasslandplant2:0.4 grasslandgrass:0.4 wheat:0.25
include stone
# Anything else that is clearly green.
ground * 0.51 +green lawn grasslandflora:0.6 grasslandplant2:0.4 grasslandgrass:0.4 wheat:0.25
