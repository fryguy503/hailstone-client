# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). The shared flora for the outdoor
# classic zones: a zone without a file of its own takes these when the server calls it outdoor. Grass grows on level,
# open ground where the texel under it is green, so sand, snow, rock, roads, water and lava stay bare. It is a lawn:
# an even carpet of blades the client builds itself (no game asset), taking the ground's colour. The first matching
# line wins.
max_slope 45
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 0 +greenish lawn
ground *grass* 0 +greenish lawn
ground *gras* 0 +greenish lawn
include stone
# Anything else that is clearly green.
ground * 0 +green lawn
